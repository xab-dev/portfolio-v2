### §1 — Rôle (remplace)

### §2 — Modèle de contenu (remplace)
```ts
export type PlaygroundCase = {
  id: 'court' | 'moyen' | 'long' | 'bizarre';
  label: string;                    // "Prompt court", "Prompt moyen", "Prompt long", "Prompt bizarre"
  rawPrompt: string;                // l'entrée brute, réaliste, fautes/hésitations comprises pour `long`
  verdict: {                        // ce que l'agent scripté "décide" — affiché AVANT le prompt expert
    kind: 'template' | 'conseil';
    templateRef?: 'T1' | 'T2' | 'T3' | 'T4' | 'T5' | 'T6';   // référence à la bibliothèque (nom affiché via un map)
    headline: string;               // 1 ligne : "→ Trop court pour une spec : 5 questions puis Cahier des charges (T4)"

  };
  expertPrompt: string;             // ≤ 25 lignes ; pour kind='conseil' : le conseil structuré, pas un prompt
  rawOutput: string;                // ce qu'un LLM aurait produit avec le brut (court, générique, plausible)

export const templateNames: Record<string, string> = {
  T1: 'Micro-Ticket', T2: 'Module Standard', T3: 'Session Diagnostic',
  T4: 'Cahier des charges technique', T5: 'Roadmap + specs numérotées', T6: 'Notes de session brutes / Triage',
};
```
